% Custom commands

%---- VERTICAL RHYTHM KNOBS ----------------------------------------
% All spacing is driven by these five lengths. Nudge these instead of
% sprinkling \vspace{-Npt} around the document; everything stays even.
\newlength{\entrygap}   \setlength{\entrygap}{2pt}   % between entries (a job, a school)
\newlength{\contgap}    \setlength{\contgap}{-5pt}   % above a continuation line of the entry above
\newlength{\contgapend} \setlength{\contgapend}{-8pt} % below one (see \resumeSubSubheading)
\newlength{\listendgap} \setlength{\listendgap}{-4pt}  % correction when a list ENDS on one
\newlength{\groupgap}   \setlength{\groupgap}{4pt}   % above a bold group label inside an entry
\newlength{\bulletgap}  \setlength{\bulletgap}{0pt}  % between bullets
\newlength{\groupindent}\setlength{\groupindent}{0.15in} % how far group labels sit left of the bullets
%--------------------------------------------------------------------

\newcommand{\resumeItem}[1]{\item \small{#1}}

% Entry heading: {title}{right-hand note}{subtitle}{right-hand note}
\newcommand{\resumeSubheading}[4]{%
  \resumecontfalse
  \vspace{\entrygap}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}%
}

% A continuation line belonging to the entry directly above (a second degree,
% a second title) — hugs the block it extends. Its 1-row table sits shallower
% than \resumeSubheading's 2-row one, so \contgapend evens out what follows.
% \ifresumecont records whether the list ended on one of these, so \contgapend
% isn't stolen from the following section heading's breathing room.
\newif\ifresumecont \resumecontfalse
\newcommand{\resumeSubSubheading}[2]{%
  \resumeconttrue
  \vspace{\contgap}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{\contgapend}%
}

\newcommand{\resumeProjectHeading}[2]{%
  \vspace{\entrygap}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \small#1 & #2 \\
    \end{tabular*}%
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}}

% Bold, un-bulleted label that divides one entry into named groups.
% Outdented to the entry's own left edge so it reads as a heading
% rather than as another bullet.
\newcommand{\resumeItemGroup}[1]{%
  \vspace{\groupgap}\item[]\hspace*{-\groupindent}\small\textbf{#1}%
}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

% Outer list: one \item per entry, no bullet.
\newcommand{\resumeSubHeadingListStart}{%
  \begin{itemize}[leftmargin=0.15in, label={},
                  itemsep=0pt, parsep=0pt, topsep=2pt, partopsep=0pt]}
\newcommand{\resumeSubHeadingListEnd}{\ifresumecont\vspace{\listendgap}\fi\end{itemize}}

% Bullets within an entry.
\newcommand{\resumeItemListStart}{%
  \begin{itemize}[leftmargin=\groupindent, itemsep=\bulletgap,
                  parsep=0pt, topsep=2pt, partopsep=0pt]}
\newcommand{\resumeItemListEnd}{\resumecontfalse\end{itemize}}

% Nested bullets — a list hanging off the bullet above it.
\newcommand{\resumeSubItemListStart}{%
  \begin{itemize}[leftmargin=0.28in, label={\textendash}, itemsep=0pt,
                  parsep=0pt, topsep=1pt, partopsep=0pt]}
\newcommand{\resumeSubItemListEnd}{\end{itemize}}
