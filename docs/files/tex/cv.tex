% Note beneath a degree line (thesis, capstone, expansion of an acronym).
% Set roman and flush with the degree title above it; the actual TITLE of a
% paper or project goes in \textit{} so italics always mean "this is a title."
\newcommand{\cvDegreeNote}[1]{%
  \vspace{-2pt}\item[]{\small#1}%
}

% Bold small-caps label dividing a section into named groups; sits at the
% section's left margin so it reads above, not alongside, the entries.
\newcommand{\cvGroupHeading}[1]{%
  \resumecontfalse
  \vspace{\groupgap}\item[]\hspace*{-\groupindent}{\small\bfseries\scshape #1}%
}

% Bold, un-bulleted label that divides one entry into named groups.
\newcommand{\resumeItemGroup}[1]{%
  \vspace{\groupgap}\item[]\hspace*{-\groupindent}\small\textbf{#1}%
}
